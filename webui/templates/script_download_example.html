FastAPI Script Download Examples

Practical implementations for triggering file downloads

Pattern 1: Simple HTML Link

The most straightforward approach. Uses standard HTML anchor tag with download attribute.

<a href="/api/scripts/download/script_123"
   download
   class="btn btn-primary">
    <i class="bi bi-download"></i> Download Script
</a>
When to use: Simple pages with static script IDs, no dynamic behavior needed

Pattern 2: JavaScript Trigger

Use JavaScript to dynamically generate download links, handle errors, or add logic.

function downloadScript(scriptId) {
    const link = document.createElement('a');
    link.href = `/api/scripts/download/${scriptId}`;
    link.download = true;  // Force download instead of navigation
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
}
When to use: Dynamic script IDs from variables, need to add click handlers, or customize behavior

Pattern 3: Fetch API with Progress

Use Fetch API for advanced control - progress tracking, error handling, custom filename extraction.

async function downloadWithFetch(scriptId) {
    try {
        const response = await fetch(`/api/scripts/download/${scriptId}`);

        if (!response.ok) {
            throw new Error(`HTTP ${response.status}: ${response.statusText}`);
        }

        // Extract filename from header
        const contentDisposition = response.headers.get('content-disposition');
        let filename = 'script.txt';
        if (contentDisposition) {
            const match = contentDisposition.match(/filename="([^"]+)"/);
            if (match) filename = match[1];
        }

        // Get file data
        const blob = await response.blob();

        // Trigger download
        const url = window.URL.createObjectURL(blob);
        const link = document.createElement('a');
        link.href = url;
        link.download = filename;
        link.click();

        window.URL.revokeObjectURL(url);

    } catch (error) {
        console.error('Download failed:', error);
        alert('Failed to download script: ' + error.message);
    }
}
When to use: Need error handling, custom filename extraction, or progress tracking

Pattern 4: Fetch with Streaming Progress

For larger files, use ReadableStream to show real-time download progress.

async function downloadWithProgress(scriptId) {
    try {
        const response = await fetch(`/api/scripts/download/${scriptId}`);
        const contentLength = response.headers.get('content-length');
        const reader = response.body.getReader();
        const chunks = [];
        let downloadedSize = 0;

        while (true) {
            const { done, value } = await reader.read();
            if (done) break;

            chunks.push(value);
            downloadedSize += value.length;

            if (contentLength) {
                const progress = (downloadedSize / contentLength) * 100;
                document.getElementById('progressBar').style.width = progress + '%';
            }
        }

        const blob = new Blob(chunks);
        const url = window.URL.createObjectURL(blob);
        const link = document.createElement('a');
        link.href = url;
        link.download = `script_${scriptId}.txt`;
        link.click();
        window.URL.revokeObjectURL(url);

    } catch (error) {
        console.error('Download failed:', error);
    }
}
When to use: Large files where progress feedback is important for user experience

Pattern 5: Form Submission

Pattern 7: Copy to Clipboard (Alternative)

Instead of downloading, copy script content directly to clipboard.

async function copyScriptToClipboard(scriptId) {
    try {
        const response = await fetch(`/api/scripts/preview/${scriptId}`);
        const data = await response.json();

        await navigator.clipboard.writeText(data.content);

        alert('Script copied to clipboard!');
    } catch (error) {
        alert('Failed to copy: ' + error.message);
    }
}
When to use: When users might want to paste directly into editor, or as complement to download
